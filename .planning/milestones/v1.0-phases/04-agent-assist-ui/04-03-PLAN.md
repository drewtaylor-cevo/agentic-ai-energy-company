Implement the hook per D-01 (native fetch, no data-fetching library), D-03 (mock fallback), D-04 (no auto-retry). The state machine has 4 states:

```typescript
import { useState, useCallback, useRef } from 'react';
import type { RecommendationResponse } from '@/lib/types';
import { normalizeCustomerId, CUSTOMER_ID_PATTERN } from '@/lib/validate';
import { MOCK_RECOMMENDATIONS } from '@/lib/mock/recommendations';

export type RecommendationState =
  | { status: 'idle' }
  | { status: 'loading' }
  | { status: 'success'; data: RecommendationResponse; customerId: string }
  | { status: 'error'; httpStatus: number; customerId: string };

export function useRecommendations() {
  const [state, setState] = useState<RecommendationState>({ status: 'idle' });
  const abortRef = useRef<AbortController | null>(null);

  const lookup = useCallback(async (rawId: string) => {
    // 1. Cancel any in-flight request (re-query clears previous per UI-SPEC)
    abortRef.current?.abort();

    // 2. Normalize + client-side validate (D-10)
    const customerId = normalizeCustomerId(rawId);
    if (!CUSTOMER_ID_PATTERN.test(customerId)) {
      setState({ status: 'error', httpStatus: 400, customerId });
      return;
    }

    // 3. Clear previous results — loading state shown immediately (UI-02)
    setState({ status: 'loading' });

    // 4. Branch: mock fallback if VITE_API_URL unset (D-03), else fetch
    const apiUrl = import.meta.env.VITE_API_URL;

    if (!apiUrl) {
      // Mock mode
      const mockData = MOCK_RECOMMENDATIONS[customerId];
      if (!mockData) {
        setState({ status: 'error', httpStatus: 404, customerId });
        return;
      }
      setState({ status: 'success', data: mockData, customerId });
      return;
    }

    // Real API fetch
    const ctrl = new AbortController();
    abortRef.current = ctrl;

    try {
      const response = await fetch(
        `${apiUrl}/recommendations/${encodeURIComponent(customerId)}`,
        { signal: ctrl.signal }
      );

      if (!response.ok) {
        setState({ status: 'error', httpStatus: response.status, customerId });
        return;
      }

      const data: RecommendationResponse = await response.json();
      setState({ status: 'success', data, customerId });
    } catch (err: unknown) {
      // Aborted requests (superseded by a new lookup) are silently ignored
      if (err instanceof DOMException && err.name === 'AbortError') {
        return;
      }
      // Network failure or other error — httpStatus 0 signals no HTTP response
      setState({ status: 'error', httpStatus: 0, customerId });
    }
  }, []);

  return { state, lookup } as const;
}
```

Key implementation notes:
- `encodeURIComponent(customerId)` in the URL prevents injection via the path segment (T-04-03 mitigation).
- Status-first parse: `response.ok` is checked before `response.json()`. Non-200 body is NOT parsed — copy is owned by UI-SPEC via `errorCopyForStatus()` in the component layer.
- No retry (D-04): a single fetch per `lookup()` call.
- AbortController for re-query: previous in-flight cancelled when new lookup starts (UI-SPEC "Re-query" state).
- Mock fallback: when `VITE_API_URL` is empty/undefined, reads from `MOCK_RECOMMENDATIONS`. Unknown IDs throw 404 so the error path is exercised in mock mode.

import { describe, it, expect, vi, beforeEach } from 'vitest';
import { renderHook, act } from '@testing-library/react';
import { useRecommendations } from './useRecommendations';

// Canonical mock data for test assertions
const MOCK_SUCCESS = {
  green:    { plan_id: 'ECO', plan_name: 'EcoFlex 100', saving_monthly: 30.00, saving_annual: 360.00 },
  cheapest: { plan_id: 'VAL', plan_name: 'Value 12',    saving_monthly: 55.00, saving_annual: 660.00 },
};

beforeEach(() => {
  vi.restoreAllMocks();
  vi.unstubAllEnvs();
});

describe('useRecommendations — API mode', () => {
  beforeEach(() => {
    vi.stubEnv('VITE_API_URL', 'https://api.example.com');
  });

  it('returns success state on HTTP 200', async () => {
    vi.stubGlobal('fetch', vi.fn(() =>
      Promise.resolve(new Response(JSON.stringify(MOCK_SUCCESS), { status: 200 }))
    ));
    const { result } = renderHook(() => useRecommendations());
    expect(result.current.state.status).toBe('idle');

    await act(async () => { await result.current.lookup('CUST-001'); });

    expect(result.current.state.status).toBe('success');
    if (result.current.state.status === 'success') {
      expect(result.current.state.data.green.saving_monthly).toBe(30.00);
      expect(result.current.state.data.cheapest.saving_monthly).toBe(55.00);
    }
  });

  it.each([
    [400, 'error'],
    [404, 'error'],
    [500, 'error'],
    [502, 'error'],
    [504, 'error'],
  ])('HTTP %i maps to error state', async (httpStatus) => {
    vi.stubGlobal('fetch', vi.fn(() =>
      Promise.resolve(new Response('{"error":"x"}', { status: httpStatus }))
    ));
    const { result } = renderHook(() => useRecommendations());
    await act(async () => { await result.current.lookup('CUST-001'); });
    expect(result.current.state.status).toBe('error');
    if (result.current.state.status === 'error') {
      expect(result.current.state.httpStatus).toBe(httpStatus);
    }
  });

  it('network failure maps to error with httpStatus 0', async () => {
    vi.stubGlobal('fetch', vi.fn(() => Promise.reject(new TypeError('Failed to fetch'))));
    const { result } = renderHook(() => useRecommendations());
    await act(async () => { await result.current.lookup('CUST-001'); });
    expect(result.current.state.status).toBe('error');
    if (result.current.state.status === 'error') {
      expect(result.current.state.httpStatus).toBe(0);
    }

  it('invalid ID (post-normalize) returns 400 without fetching', async () => {

    if (result.current.state.status === 'error') {
