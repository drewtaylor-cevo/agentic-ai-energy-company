      echo "FATAL: lockfiles have uncommitted changes:"

      exit 1

    # Expected: exit 0, no ERROR lines

    # 4. CDK synth all 3 stacks offline
    AWS_DEFAULT_REGION=us-east-1 npx aws-cdk@latest synth --all --quiet > /tmp/phase5-synth-stdout.txt 2> /tmp/phase5-synth-stderr.txt

    pytest -m "not smoke" tests/ -x -q > /tmp/phase5-pytest-offline.txt 2>&1

    # 6. Cleanup the temporary venv
    rm -rf .venv-lock

    # 7. Confirm ui/.gitignore is unchanged and dist is still ignored
    test -z "$(git status --porcelain ui/.gitignore)" || { echo "FATAL: ui/.gitignore unexpectedly modified"; exit 1; }

    pytest -m "not smoke" exit: 0

    Captured: $(date -u +%Y-%m-%dT%H:%M:%SZ)
