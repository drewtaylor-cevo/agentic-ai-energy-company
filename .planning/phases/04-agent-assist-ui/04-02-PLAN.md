Reproduce the backend regex from `api_lambda/handler.py:27` byte-for-byte, plus D-10 normalization rules:

```typescript
// Must match api_lambda/handler.py:27 and lambda/handler.py:39 exactly.
export const CUSTOMER_ID_PATTERN = /^CUST-\d{3,6}$/;

// Auto-insert dash pattern: matches CUST followed by 3-6 digits with no dash.
const DASHLESS_PATTERN = /^CUST(\d{3,6})$/;

/**
 * Normalize a customer ID input per D-10:
 * 1. Trim whitespace
 * 2. Uppercase
 * 3. Auto-insert dash if operator typed CUSTnnnnn (no dash)
 */
export function normalizeCustomerId(raw: string): string {
  const trimmed = raw.trim().toUpperCase();
  const dashless = DASHLESS_PATTERN.exec(trimmed);
  if (dashless) {
    return `CUST-${dashless[1]}`;
  }
  return trimmed;
}
```

```typescript
/**
 * Returns operator-facing error copy keyed by HTTP status code.
 * Copy strings from UI-SPEC §Copywriting Contract (lines 119-122).
 * The UI IGNORES response.json().error text — it keys by status code only.
 */
export function errorCopyForStatus(status: number, customerId: string): string {
  switch (status) {
    case 400:
      return "That doesn't look like a customer ID. Format is CUST followed by 3–6 digits.";
    case 404:
      return `No customer found for ${customerId}. Check the ID and try again.`;
    case 504:
      return "Recommendations are taking longer than expected. Try again in a moment.";
    case 500:
    case 502:
      return "Something went wrong on our end. Try again — if it persists, contact support.";
    default:
      return "Something went wrong on our end. Try again — if it persists, contact support.";
  }
}
```
