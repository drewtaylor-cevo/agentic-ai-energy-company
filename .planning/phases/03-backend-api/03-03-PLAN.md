```python
"""Live HTTP smoke tests for the Phase 3 backend API.

Run AFTER `cdk deploy CustomerTariffApi` succeeds:

    export BACKEND_API_URL=$(aws cloudformation describe-stacks \
        --stack-name CustomerTariffApi --region us-east-1 \

Tests are SKIPPED if BACKEND_API_URL is not set, so they do not break
local CI for developers without a deployed stack (mirrors the Phase 2
`tests/test_agent_smoke.py` AGENT_RUNTIME_ARN pattern).

Covers Phase 3 success criteria:
  SC-1: curl to endpoint with any demo persona returns {green, cheapest}
        — no live CRM, no external data source required (DEMO-01)
  SC-2: Error cases handled gracefully — 400 bad format, 404 unknown
        customer, 504 agent timeout — each with a friendly {"error": "..."}
        body, never a stack trace (D-12)
  SC-3: Fresh session ID per invocation — no recommendation bleed between
        consecutive persona lookups (D-11). Verified here via the proxy
        "different-persona-different-savings"; unit-level uuid4 isolation
        is proven in tests/test_backend_api_handler.py::test_fresh_session_id_per_call.
"""
import os

import pytest
import requests

# rstrip("/") defends against CDK's api.url trailing-slash semantics — the
# URL construction f"{BACKEND_API_URL}/recommendations/..." is clean either way.
BACKEND_API_URL = os.environ.get("BACKEND_API_URL", "").rstrip("/")

pytestmark = [
    pytest.mark.smoke,
    pytest.mark.skipif(
        not BACKEND_API_URL,
        reason="BACKEND_API_URL not set — skip live backend API smoke tests",
    ),
]

# Generous timeout: first invocation may hit a cold start on both the
# API Lambda and the AgentCore runtime container. D-04 accepts cold-start
# latency for the demo; the 60s here comfortably exceeds the 30s Lambda
# timeout + API Gateway 29s integration limit.
_HTTP_TIMEOUT = 60


def _reco_url(customer_id: str) -> str:
    return f"{BACKEND_API_URL}/recommendations/{customer_id}"


# --- SC-1: Positive path for all 3 demo personas (DEMO-01 literal) ---

@pytest.mark.parametrize("customer_id", ["CUST-001", "CUST-002", "CUST-003"])
def test_all_personas_return_recommendations(customer_id):
    """SC-1: every demo persona returns HTTP 200 + {green, cheapest} body.

    This is the literal "curl to endpoint returns recommendations" check
    from Phase 3 success criterion 1 (ROADMAP.md).
    """
    r = requests.get(_reco_url(customer_id), timeout=_HTTP_TIMEOUT)
    assert r.status_code == 200, (
        f"{customer_id} expected 200, got {r.status_code}: {r.text[:300]}"
    )
    body = r.json()
    # D-02 pass-through: body has exactly green + cheapest at the top level
    assert "green" in body, f"{customer_id} response missing 'green' track"
    assert "cheapest" in body, f"{customer_id} response missing 'cheapest' track"
    # Each track carries the 4 RecommendationResponse fields (agent/agent.py)
```
