```python
"""Phase 3 construct — Backend API Lambda + API Gateway HTTP API v2.

Builds the API Lambda (api_lambda/ asset, Python 3.12, 30s timeout per D-03),
a HTTP API v2 with allow-all CORS (D-09), the route GET /recommendations/{customer_id}
(D-10) bound to the Lambda via HttpLambdaIntegration, and an IAM policy scoped
to bedrock-agentcore:InvokeAgentRuntime on the specific runtime ARN (RESEARCH.md Q6).

The AgentCore runtime ARN is passed in as a kwarg (CDK token from
ssm.StringParameter.value_for_string_parameter in BackendApiStack). CloudFormation
resolves the token at deploy time — no AWS creds required at synth (Pattern 5).
"""
import aws_cdk as cdk
from aws_cdk import Duration
from aws_cdk import aws_apigatewayv2 as apigwv2
from aws_cdk import aws_apigatewayv2_integrations as integ
from aws_cdk import aws_iam as iam
from aws_cdk import aws_lambda as lambda_
from constructs import Construct


class BackendApiConstruct(Construct):
    """API Lambda fronted by a HTTP API v2 with allow-all CORS."""

    def __init__(
        self,
        scope: Construct,
        construct_id: str,
        *,
        agent_runtime_arn: str,
    ) -> None:
        super().__init__(scope, construct_id)

        stack = cdk.Stack.of(self)

        # Lambda function — api_lambda/ asset, Python 3.12, 30s timeout (D-03).
        # handler="handler.handler" refers to api_lambda/handler.py :: def handler
        # (RESEARCH.md Pitfall 7 — module.function mismatch is a common trap).
        fn = lambda_.Function(
            self,
            "TariffApiLambda",
            function_name="tariff-api",
            runtime=lambda_.Runtime.PYTHON_3_12,   # Match Phase 1 ToolsLambda
            handler="handler.handler",
            code=lambda_.Code.from_asset("api_lambda"),
            timeout=Duration.seconds(30),          # D-03: Lambda 30s aligned with HTTP API v2 hard limit
            memory_size=256,                       # Match Phase 1 default
            environment={
                "AGENT_RUNTIME_ARN": agent_runtime_arn,
                "AWS_REGION": stack.region,
                "LOG_LEVEL": "INFO",
            },
            description="Phase 3: proxies GET /recommendations/{customer_id} to AgentCore",
        )

        # IAM — scoped to InvokeAgentRuntime on the specific runtime ARN (Q6).
        # No grant_* helper exists for bedrock-agentcore; hand-write the
        # PolicyStatement (mirrors agent_runtime.py lines 67-74 scoping style).
        fn.add_to_role_policy(
            iam.PolicyStatement(
                effect=iam.Effect.ALLOW,
                actions=["bedrock-agentcore:InvokeAgentRuntime"],
                resources=[agent_runtime_arn],
            )
        )

        # HTTP API v2 — stable aws-cdk-lib module, not alpha. CORS allow-all
        # (D-09) is built into the HttpApi constructor; no explicit OPTIONS
        # route needed (API Gateway handles preflight when cors_preflight is
        # set — RESEARCH.md Pattern 1, Q10, Pitfall 4).
        api = apigwv2.HttpApi(
            self,
            "TariffApi",
            api_name="customer-tariff-api",
            description="Phase 3: customer tariff recommendation API",
            cors_preflight=apigwv2.CorsPreflightOptions(
                allow_origins=["*"],
                allow_methods=[
                    apigwv2.CorsHttpMethod.GET,
                    apigwv2.CorsHttpMethod.OPTIONS,
                ],
                allow_headers=["Content-Type"],
            ),
        )

        # Route binding — GET /recommendations/{customer_id} -> API Lambda (D-10).
        # HttpLambdaIntegration defaults to PayloadFormatVersion.VERSION_2_0,
        # which means event["pathParameters"]["customer_id"] in the handler.
        api.add_routes(
            path="/recommendations/{customer_id}",
            methods=[apigwv2.HttpMethod.GET],
            integration=integ.HttpLambdaIntegration("RecoIntegration", fn),
        )

        self._fn = fn
        self._api = api
        self._api_endpoint = api.url  # Includes trailing slash

    @property
    def api_endpoint(self) -> str:
        """HTTPS URL of the API including trailing slash."""
        return self._api_endpoint

    @property
    def function(self) -> lambda_.Function:
        return self._fn
```

```python
"""Phase 3 CDK stack — Backend API Lambda + HTTP API v2.

Reads AgentRuntime ARN from SSM (written by AgentCoreStack amendment in
Plan 02 Task 2.1) to avoid hard CloudFormation export dependencies
between stacks (Pitfall 5 / D-07). Pairs with BackendApiConstruct which
contains the Lambda + HTTP API + IAM + route wiring.
"""
from aws_cdk import CfnOutput, Stack
from aws_cdk import aws_ssm as ssm
from constructs import Construct

from infrastructure.constructs.backend_api import BackendApiConstruct


class BackendApiStack(Stack):
    def __init__(self, scope: Construct, construct_id: str, **kwargs) -> None:
        super().__init__(scope, construct_id, **kwargs)

        # value_for_string_parameter returns a CloudFormation dynamic reference
        # token that CloudFormation resolves at deploy time. No AWS creds
        # required at synth (Pattern 5). Do NOT use value_from_lookup (synth-time
        # resolution — requires creds) or Fn.import_value (export lock — Pitfall 3).
        agent_runtime_arn = ssm.StringParameter.value_for_string_parameter(
            self, "/customer-tariff/agent-runtime-arn"
        )

        backend = BackendApiConstruct(
            self,
            "BackendApi",
            agent_runtime_arn=agent_runtime_arn,
        )

        CfnOutput(self, "ApiEndpoint", value=backend.api_endpoint)
```
